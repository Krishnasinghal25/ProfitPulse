# ============================================================
# OBJECTIVE 12
# Problem statement: Encode and assemble model features.
# ============================================================

heading(
    12,
    "Encode categorical variables and combine all inputs "
    "into a feature vector using a Spark ML pipeline."
)

indexers = [
    StringIndexer(
        inputCol=c,
        outputCol=c + "_index",
        handleInvalid="keep"
    )
    for c in categorical_features
]

encoder = OneHotEncoder(
    inputCols=[
        c + "_index" for c in categorical_features
    ],
    outputCols=[
        c + "_encoded" for c in categorical_features
    ],
    handleInvalid="keep"
)

assembler = VectorAssembler(
    inputCols=(
        numeric_features
        + [c + "_encoded" for c in categorical_features]
    ),
    outputCol="features"
)

preprocessing_pipeline = Pipeline(
    stages=indexers + [encoder, assembler]
)

# Fit preprocessing only on training data.
preprocessing_model = preprocessing_pipeline.fit(train_df)

train_ready = (
    preprocessing_model.transform(train_df)
    .select("features", "label")
    .cache()
)

test_ready = (
    preprocessing_model.transform(test_df)
    .select("features", "label")
    .cache()
)

train_ready.show(5, truncate=False)

vector_size = (
    train_ready.select("features").first()["features"].size
)

conclusion(
    "StringIndexer and OneHotEncoder encoded the categorical "
    "variables. VectorAssembler combined the inputs into "
    f"a feature vector with {vector_size} positions. "
    "Preprocessing was fitted only on the training set "
    "to prevent information leakage from the test set."
)


# ============================================================
# COMMON MODEL EVALUATION HELPERS
# ============================================================

accuracy_evaluator = MulticlassClassificationEvaluator(
    labelCol="label",
    predictionCol="prediction",
    metricName="accuracy"
)

roc_evaluator = BinaryClassificationEvaluator(
    labelCol="label",
    rawPredictionCol="probability",
    metricName="areaUnderROC"
)

pr_evaluator = BinaryClassificationEvaluator(
    labelCol="label",
    rawPredictionCol="probability",
    metricName="areaUnderPR"
)


def get_metrics(name, predictions):
    counts = {
        (int(row["label"]), int(row["prediction"])): row["count"]
        for row in predictions.groupBy(
            "label", "prediction"
        ).count().collect()
    }

    tn = counts.get((0, 0), 0)
    fp = counts.get((0, 1), 0)
    fn = counts.get((1, 0), 0)
    tp = counts.get((1, 1), 0)

    precision = tp / (tp + fp) if tp + fp else 0.0
    recall = tp / (tp + fn) if tp + fn else 0.0

    f1 = (
        2 * precision * recall / (precision + recall)
        if precision + recall else 0.0
    )

    return {
        "model": name,
        "accuracy": accuracy_evaluator.evaluate(predictions),
        "roc_auc": roc_evaluator.evaluate(predictions),
        "pr_auc": pr_evaluator.evaluate(predictions),
        "yes_precision": precision,
        "yes_recall": recall,
        "yes_f1": f1,
        "TN": tn,
        "FP": fp,
        "FN": fn,
        "TP": tp
    }

    print("\nConfusion matrix:", metrics["model"])

    metrics_list,

conclusion(
    f"{best_recall['model']} has the highest subscriber "
    f"recall among the compared models at "
    f"{best_recall['yes_recall'] * 100:.2f}%. "
    f"{best_auc['model']} has the highest ROC-AUC "
    f"at {best_auc['roc_auc']:.4f}. "
    f"Always predicting 'no' already gives "
    f"{baseline_accuracy * 100:.2f}% accuracy, "
